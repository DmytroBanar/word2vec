% Знайти рівняння площини, яка буде паралельна вказаній вісі і проходитиме через вказані точки
A = [2, -3, 2];
B = [7, 1, 0];

% Вектор AB
AB = B - A;

% Вектор, паралельний осі Ox
vx = [1, 0, 0];

% Нормальний вектор площини
n = cross(AB, vx);

% Компоненти нормального вектора
n1 = n(1);
n2 = n(2);
n3 = n(3);

% Координати точки A
x0 = A(1);
y0 = A(2);
z0 = A(3);

% Створення символічного рівняння площини
syms x y z
eq = n1*(x - x0) + n2*(y - y0) + n3*(z - z0);
eq = simplify(eq);

disp('Рівняння площини:');
pretty(eq)

% Побудова графіка площини, що проходить через A і B, перпендикулярна Ox
figure;
hold on;
grid on;
xlabel('X');
ylabel('Y');
zlabel('Z');
title('Площина через A і B, перпендикулярна осі Ox');
axis equal;

% Візуалізація векторів і точок
plot3([A(1), B(1)], [A(2), B(2)], [A(3), B(3)], 'b', 'LineWidth', 2); % відрізок AB
scatter3(A(1), A(2), A(3), 100, 'r', 'filled');
text(A(1), A(2), A(3), 'A', 'VerticalAlignment', 'bottom', 'HorizontalAlignment', 'right');
scatter3(B(1), B(2), B(3), 100, 'g', 'filled');
text(B(1), B(2), B(3), 'B', 'VerticalAlignment', 'bottom', 'HorizontalAlignment', 'right');

% Побудова площини
[xp, yp] = meshgrid(x0-3:0.5:x0+3, y0-3:0.5:y0+3);

% Обчислення z з рівняння площини: n1*(x - x0) + n2*(y - y0) + n3*(z - z0) = 0
% -> z = ( -n1*(x - x0) - n2*(y - y0) ) / n3 + z0
zp = ( -n1*(xp - x0) - n2*(yp - y0) ) / n3 + z0;

surf(xp, yp, zp, 'FaceAlpha', 0.5, 'EdgeColor', 'none', 'FaceColor', [0.6 0.8 1]);

legend('Відрізок AB', 'Точка A', 'Точка B', 'Площина');
view(3);


% =================ЗАВДАННЯ 2 =================
% Значення параметра t
t_vals = [0, pi/2, pi, 3*pi/2];

% Ініціалізація масиву для зберігання результатів
points = zeros(length(t_vals), 3); % стовпці: x, y, z

% Обчислення координат точок перетину
for i = 1:length(t_vals)
    t = t_vals(i);
    
    % Еліпс: параметричне задання
    x = 2 * cos(t);
    y = 6 * sin(t);
    
    % z з площини: z = (-2x + y - 3)/3
    z = (-2*x + y - 3) / 3;
    
    % Зберегти координати
    points(i, :) = [x, y, z];
end

% Вивід результату
disp('Точки перетину еліпса та площини:');
for i = 1:size(points, 1)
    fprintf('P%d = (%.2f,\t%.2f,\t%.2f)\n', i, points(i,1), points(i,2), points(i,3));
end

% Побудова графіка
figure;
hold on;
grid on;
axis equal;
xlabel('X');
ylabel('Y');
zlabel('Z');
title('Перетин еліпса з площиною');

% Побудова еліпса (розширено для кращої візуалізації)
t = linspace(0, 2*pi, 200);
x_ellipse = 2 * cos(t);
y_ellipse = 6 * sin(t);
z_ellipse = (-2*x_ellipse + y_ellipse - 3) / 3;
plot3(x_ellipse, y_ellipse, z_ellipse, 'b', 'LineWidth', 2);

% Побудова площини
[xp, yp] = meshgrid(-3:0.5:3, -7:0.5:7);
zp = (-2*xp + yp - 3)/3;
surf(xp, yp, zp, 'FaceAlpha', 0.5, 'EdgeColor', 'none', 'FaceColor', [0.8 0.6 0.6]);

% Нанесення точок перетину
scatter3(points(:,1), points(:,2), points(:,3), 80, 'r', 'filled');

% Підписи до точок
for i = 1:size(points, 1)
    text(points(i,1), points(i,2), points(i,3), sprintf('P%d', i), ...
        'VerticalAlignment', 'bottom', 'HorizontalAlignment', 'right', 'FontSize', 10);
end

legend('Еліпс', 'Площина', 'Точки перетину');
view(3);


% ========================= ЗАВДАННЯ 3 ======================
% Частина 1: x ∈ [0, 4/3], y ∈ [0, 2x]
vol1 = integral2(@(x, y) 16 - x.^2, ...
    0, 4/3, ...
    @(x) zeros(size(x)), @(x) 2.*x);

% Частина 2: x ∈ [4/3, 4], y ∈ [0, 4 - x]
vol2 = integral2(@(x, y) 16 - x.^2, ...
    4/3, 4, ...
    @(x) zeros(size(x)), @(x) 4 - x);

% Повний об'єм
V = vol1 + vol2;

% Вивід результату
fprintf('Об’єм тіла: %.4f одиниць кубічних\n', V);
